\section{Learnable Fourier Time Encoder Module}
\begin{lstlisting}[language=python, caption={PyTorch Implementation of LearnableFourierTimeEncoder in Model Architecture}]
import math
import torch
import torch.nn as nn

class LearnableFourierTimeEncoder(nn.Module):
    """Continuous-Time Learnable Fourier Harmonic Encoder.
    Projects scalar time deltas Delta_t into a 2*d dimensional vector
    using learnable frequencies (omega) and phase shifts (phi).
    """
    def __init__(self, time_dim: int = 64):
        super().__init__()
        self.time_dim = time_dim
        # Initialise frequencies with geometric progression
        init_freq = 1.0 / (10000.0 ** (torch.arange(0, time_dim).float() / time_dim))
        self.omega = nn.Parameter(init_freq)
        self.phi = nn.Parameter(torch.zeros(time_dim))

    def forward(self, delta_t: torch.Tensor) -> torch.Tensor:
        # delta_t shape: [batch_size, num_neighbors] or [num_edges]
        delta_t = delta_t.unsqueeze(-1)  # [..., 1]
        angles = delta_t * self.omega + self.phi  # [..., time_dim]
        # Return concatenated cosine and sine projections
        fourier_enc = torch.cat([torch.cos(angles), torch.sin(angles)], dim=-1)
        return fourier_enc / math.sqrt(self.time_dim)
\end{lstlisting}

\section{Causal Temporal Neighbor Sampler}
\begin{lstlisting}[language=python, caption={Causal Temporal Neighbor Sampler Enforcing Historical Precedence}]
class TemporalNeighborSampler:
    """Samples up to M causal historical neighbors strictly where t_e <= t_target."""
    def __init__(self, edge_index: torch.Tensor, edge_times: torch.Tensor, max_neighbors: int = 20):
        self.max_neighbors = max_neighbors
        self.adj = {}
        # Pre-build chronological adjacency list
        for (src, dst), t in zip(edge_index.t().tolist(), edge_times.tolist()):
            if dst not in self.adj:
                self.adj[dst] = []
            self.adj[dst].append((src, t))
        # Sort each node's neighbors descending by timestamp
        for dst in self.adj:
            self.adj[dst].sort(key=lambda x: x[1], reverse=True)

    def sample(self, node_id: int, target_time: float):
        neighbors = self.adj.get(node_id, [])
        # Strict causal filter: t_neighbor <= target_time
        causal = [n for n in neighbors if n[1] <= target_time]
        sampled = causal[:self.max_neighbors]
        return sampled
\end{lstlisting}
